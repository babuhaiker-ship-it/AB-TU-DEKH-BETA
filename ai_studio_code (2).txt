# --- Force Subscribe Check ---
async def check_membership(client: Client, user_id: int) -> bool:
    if is_admin(user_id):
        return True

    if not FORCE_SUB_CHANNELS:
        logger.info("No force subscribe channels configured. Skipping membership check.")
        return True

    async def check_channel(channel_id):
        try:
            member = await client.get_chat_member(channel_id, user_id)
            return member.status in [ChatMemberStatus.MEMBER, ChatMemberStatus.ADMINISTRATOR, ChatMemberStatus.OWNER]
        except PeerIdInvalid:
            logger.error(f"Force subscribe channel ID {channel_id} is invalid. Please check config.")
            return True
        except Exception as e:
            logger.error(f"Error checking membership for user {user_id} in channel {channel_id}: {e}", exc_info=True)
            return False

    results = await asyncio.gather(*[check_channel(c['channel_id']) for c in FORCE_SUB_CHANNELS])
    is_member_of_all = all(results)
    if not is_member_of_all:
        logger.info(f"User {user_id} is NOT a member of all force sub channels. Results: {results}")

    return is_member_of_all

# --- Utility Functions ---
def str_to_b64(string: str) -> str:
    return base64.urlsafe_b64encode(string.encode('utf-8')).decode('utf-8')

def b64_to_str(b64: str) -> str:
    try:
        return base64.urlsafe_b64decode(b64.encode('utf-8')).decode('utf-8')
    except Exception as e:
        logger.error(f"Failed to decode base64 string: {e}")
        return ""

def get_current_time() -> int:

async def get_shortener_config_and_shorten_url(long_url: str) -> str:
    logger.info(f"Attempting to shorten URL: {long_url}")
    shortener_config = settings_collection.find_one({'_id': 'shortener_config'})
    if not shortener_config:
        logger.warning("URL shortener configuration not found in DB. Returning original URL.")
        return long_url

    template_url = shortener_config.get('template_url')
    if not template_url or '{long_url}' not in template_url:
        logger.warning("Missing or invalid 'template_url' in shortener config. Returning original URL.")
        return long_url

    try:
        parsed_url = urllib.parse.urlparse(template_url)
        base_site = parsed_url.netloc
        query_params = urllib.parse.parse_qs(parsed_url.query)
        api_key_list = query_params.get('api')
        if not api_key_list:
            logger.warning("No 'api' key found in the template_url. Returning original URL.")
            return long_url
        api_key = api_key_list[0]
        base_site = parsed_url.netloc
        shortzy_client = Shortzy(api_key=api_key, base_site=base_site)
        shortened_url = await shortzy_client.convert(long_url)
        if shortened_url and (shortened_url.startswith("http://") or shortened_url.startswith("https://")):
            logger.info(f"URL shortened successfully to: {shortened_url}")
            return shortened_url
        else:
            logger.warning(f"Shortzy returned an invalid string: {shortened_url}. Returning original URL.")
            return long_url
    except Exception as e:
        logger.error(f"Error using Shortzy to shorten URL {long_url}: {e}", exc_info=True)
        return long_url

def decode_file_id(file_id: str):
    decoded = FileId.decode(file_id)
    file_type = decoded.file_type
    if file_type == FileType.PHOTO:
        return InputPhotoFileLocation(
            id=decoded.media_id, access_hash=decoded.access_hash,
            file_reference=decoded.file_reference, thumb_size=decoded.thumbnail_file_type
        ), decoded.file_size, decoded.dc_id, decoded.file_type, decoded.file_name
    elif file_type in [FileType.DOCUMENT, FileType.VIDEO, FileType.ANIMATION, FileType.VOICE, FileType.AUDIO]:
        return InputDocumentFileLocation(
            id=decoded.media_id, access_hash=decoded.access_hash,
            file_reference=decoded.file_reference, thumb_size=""
        ), decoded.file_size, decoded.dc_id, decoded.file_type, decoded.file_name
    else:
        return None, None, None, None, None

def get_mime_type(file_name: str = None, file_type: FileType = None) -> str:
    if file_name:
        mime, _ = mimetypes.guess_type(file_name)
        if mime: return mime
    if file_type == FileType.VIDEO: return "video/mp4"
    if file_type == FileType.AUDIO: return "audio/mpeg"
    return "application/octet-stream"

# --- Token and Premium Management ---
def add_token(user_id: int, duration_seconds: int = config.TOKEN_EXPIRY, is_admin_granted: bool = False):
    now = datetime.utcnow()
    expires_at = now + timedelta(seconds=duration_seconds)
    token = {'token_id': str(uuid.uuid4()), 'created_at': now, 'expires_at': expires_at, 'is_admin_granted': is_admin_granted}
    try:
        tokens_collection.update_one({'user_id': user_id}, {'$push': {'tokens': token}}, upsert=True)
        logger.info(f"Token added for user {user_id}. Admin granted: {is_admin_granted}, expires at {expires_at}")
        return token
    except Exception as e:
        logger.error(f"Error adding token for user {user_id}: {e}", exc_info=True)
        return None

def is_premium_user(user_id: int) -> bool:
    now = datetime.utcnow()
    doc = tokens_collection.find_one({'user_id': user_id})
    if not doc or 'tokens' not in doc: return False
    return any(token.get('is_admin_granted', False) and token.get('expires_at') and token['expires_at'] > now for token in doc['tokens'])

def user_has_token(user_id: int) -> bool:
    now = datetime.utcnow()
    doc = tokens_collection.find_one({'user_id': user_id})
    if not doc or 'tokens' not in doc: return False
    return any(token.get('expires_at') and token['expires_at'] > now for token in doc['tokens'])

async def check_and_update_free_batch_usage(user_id: int, batch_id: str) -> bool:
    if is_premium_user(user_id) or user_has_token(user_id):
        return True
    now = datetime.utcnow()
    user_doc = users_collection.find_one_and_update(
        {'user_id': user_id},
        {'$setOnInsert': {'free_batch_usage': {'claimed_batches': [], 'reset_at': now + timedelta(hours=config.FREE_LIMIT_RESET_HOURS)}}},
        upsert=True, return_document=ReturnDocument.AFTER
    )
    free_usage = user_doc.get('free_batch_usage', {})
    claimed_batches = free_usage.get('claimed_batches', [])
    reset_at = free_usage.get('reset_at')
    if not reset_at or now >= reset_at:
        new_reset_at = now + timedelta(hours=config.FREE_LIMIT_RESET_HOURS)
        users_collection.update_one({'user_id': user_id}, {'$set': {'free_batch_usage': {'claimed_batches': [], 'reset_at': new_reset_at}}})
        claimed_batches = []
        logger.info(f"Reset free batch limit for user {user_id}. Next reset at {new_reset_at}.")
    if batch_id in claimed_batches:
        logger.info(f"User {user_id} has access to already claimed batch {batch_id}.")
        return True
    if len(claimed_batches) < config.FREE_BATCH_LIMIT:
        users_collection.update_one({'user_id': user_id}, {'$push': {'free_batch_usage.claimed_batches': batch_id}})
        logger.info(f"User {user_id} claimed a new free batch {batch_id}. Total claimed today: {len(claimed_batches) + 1}/{config.FREE_BATCH_LIMIT}.")
        return True
    else:
        logger.warning(f"User {user_id} has reached their free batch limit for today.")
        return False

# --- Referral System ---
def handle_referral(new_user_id: int, ref_code: str) -> int | None:
    if ref_code.startswith('ref_'):
        try:
            referrer_id = int(ref_code[4:])
            if referrer_id != new_user_id and users_collection.find_one({'user_id': referrer_id}):
                add_token(referrer_id, config.REFERRAL_BONUS * 86400, is_admin_granted=False)
                users_collection.update_one({'user_id': referrer_id}, {'$inc': {'referral_count': 1}}, upsert=True)
                users_collection.update_one({'user_id': new_user_id}, {'$set': {'referred_by': referrer_id}}, upsert=True)
                logger.info(f"User {new_user_id} successfully referred by {referrer_id}.")
                return referrer_id
        except (ValueError, IndexError):
            logger.warning(f"Invalid referrer ID format in ref_code: {ref_code}")
    elif ref_code.startswith('video_'):
        try:
            referrer_id = int(ref_code.split('_')[2])
            if referrer_id != new_user_id and users_collection.find_one({'user_id': referrer_id}):
                add_token(referrer_id, config.REFERRAL_BONUS * 86400, is_admin_granted=False)
                users_collection.update_one({'user_id': referrer_id}, {'$inc': {'referral_count': 1}}, upsert=True)
                users_collection.update_one({'user_id': new_user_id}, {'$set': {'referred_by': referrer_id}}, upsert=True)
                logger.info(f"User {new_user_id} successfully referred by {referrer_id} via video share.")
                return referrer_id
        except (ValueError, IndexError):
            logger.warning(f"Invalid video share ref_code format: {ref_code}")
    return None

# --- Category Management ---
def validate_category_name(name: str) -> tuple[bool, str]:
    if not name: return False, "Category name cannot be empty."
    if len(name) > 64: return False, "Category name cannot be longer than 64 characters."
    return True, ""

def get_categories() -> list[str]:
    return sorted([c['name'] for c in categories_collection.find({})])

def add_category(name: str) -> tuple[bool, str]:
    try:
        valid, error = validate_category_name(name)
        if not valid: return False, error
        if categories_collection.find_one({'name': name}):
            return False, f"Category '{html.escape(name)}' already exists."
        categories_collection.insert_one({'name': name, 'created_at': datetime.utcnow()})
        logger.info(f"Category '{name}' added")
        return True, f"Category '{html.escape(name)}' added successfully."
    except Exception as e:
        logger.error(f"Error adding category '{name}': {e}")
        return False, "Failed to add category. Please try again."

def delete_category(name: str) -> tuple[bool, str, int]:
    try:
        if not categories_collection.find_one({'name': name}):
            return False, f"Category '{html.escape(name)}' does not exist.", 0
        result = media_collection.delete_many({'category': name})
        deleted_count = result.deleted_count
        categories_collection.delete_one({'name': name})
        logger.info(f"Category '{name}' and {deleted_count} videos deleted")
        return True, f"Category '{html.escape(name)}' and {deleted_count} videos deleted.", deleted_count
    except Exception as e:
        logger.error(f"Error deleting category '{name}': {e}")
        return False, "Failed to delete category. Please try again.", 0

def get_video_and_position(video_uuid: str, category: str, is_saved: bool, user_id: int = None) -> tuple[dict | None, int, int]:
    if is_saved and user_id is not None:
        user_doc = users_collection.find_one({'user_id': user_id})
        bookmarked_videos = user_doc.get('bookmarked_videos', [])
        filtered_saved_videos = [bm for bm in bookmarked_videos if (video_data := get_video_by_uuid(bm['uuid'])) and video_data.get('category') == category]
        filtered_saved_videos.sort(key=lambda x: x.get('bookmarked_at', datetime.min))
        total_videos = len(filtered_saved_videos)
        if not total_videos: return None, 0, 0
        try:
            current_video_index = [entry['uuid'] for entry in filtered_saved_videos].index(video_uuid)
            video = get_video_by_uuid(video_uuid)
            return video, current_video_index + 1, total_videos
        except ValueError:
            return None, 0, 0
    else:
        all_videos_in_category = list(media_collection.find({'category': category}, sort=[('sequence_number', ASCENDING)]))
        total_videos = len(all_videos_in_category)
        if not total_videos: return None, 0, 0
        try:
            current_video_index = [video_doc['uuid'] for video_doc in all_videos_in_category].index(video_uuid)
            video = all_videos_in_category[current_video_index]
            return video, current_video_index + 1, total_videos
        except ValueError:
            return None, 0, 0

    total_videos = len(video_uuids)
    if not (0 <= current_index < total_videos): return None, 0, 0
    video_uuid = video_uuids[current_index]
    video = get_video_by_uuid(video_uuid)
    return video, current_index + 1, total_videos

def get_next_video_chronological(current_uuid: str, category: str) -> dict | None:
    current_video = media_collection.find_one({'uuid': current_uuid, 'category': category})
    if not current_video or 'sequence_number' not in current_video:
        return get_first_video_by_sequence_number(category)
    current_sequence = current_video['sequence_number']
    next_video = media_collection.find_one({'category': category, 'sequence_number': {'$gt': current_sequence}}, sort=[('sequence_number', ASCENDING)])
    return next_video or get_first_video_by_sequence_number(category)

def get_previous_video_chronological(current_uuid: str, category: str) -> dict | None:

    return get_video_by_uuid(filtered_saved_videos[next_index]['uuid'])

def rearrange_sequences_after_deletion(category: str, deleted_sequence_number: int):
    logger.info(f"Rearranging sequences for category '{category}' after deleting sequence {deleted_sequence_number}.")
    try:
        videos_to_update = list(media_collection.find({'category': category, 'sequence_number': {'$gt': deleted_sequence_number}}).sort('sequence_number', ASCENDING))
        if not videos_to_update: return
        operations = [UpdateOne({'_id': video_doc['_id']}, {'$set': {'sequence_number': video_doc['sequence_number'] - 1}}) for video_doc in videos_to_update]
        if operations:
            result = media_collection.bulk_write(operations)
            logger.info(f"Successfully rearranged sequences for category '{category}'. Modified {result.modified_count} documents.")
    except Exception as e:
        logger.error(f"Error rearranging sequences for category '{category}': {e}", exc_info=True)

async def delete_broken_video_from_db_and_channel(client: Client, video_uuid: str, category: str, sequence_number: int, message_id_in_channel: int):
    logger.info(f"Attempting to delete broken video {video_uuid} from DB and channel.")
    if not DATA_CHANNEL_ID:
        logger.error("DATA_CHANNEL_ID is not set. Cannot delete from channel.")
        return False
    try:
        result = media_collection.delete_one({'uuid': video_uuid})
        if result.deleted_count > 0:
            logger.info(f"Successfully deleted video {video_uuid} from media_collection.")
            if category and sequence_number is not None:
                rearrange_sequences_after_deletion(category, sequence_number)
            users_collection.update_many({}, {'$pull': {'bookmarked_videos': {'uuid': video_uuid}}})
            history_collection.update_many({}, {'$pull': {'history': {'video_uuid': video_uuid}}})
            all_users_with_last_viewed = users_collection.find({'last_viewed_per_category': {'$exists': True}})
            for user_doc in all_users_with_last_viewed:
                if any(vid_uuid_in_map == video_uuid for vid_uuid_in_map in user_doc['last_viewed_per_category'].values()):
                    updated_last_viewed = {cat: vid for cat, vid in user_doc['last_viewed_per_category'].items() if vid != video_uuid}
                    users_collection.update_one({'user_id': user_doc['user_id']}, {'$set': {'last_viewed_per_category': updated_last_viewed}})
            if message_id_in_channel:
                try:
                    me = await client.get_me()
                    member = await client.get_chat_member(DATA_CHANNEL_ID, me.id)
                    if member.status in [ChatMemberStatus.ADMINISTRATOR, ChatMemberStatus.OWNER] and member.can_delete_messages:
                        await client.delete_messages(DATA_CHANNEL_ID, message_id_in_channel)
                        logger.info(f"Successfully deleted message {message_id_in_channel} from channel {DATA_CHANNEL_ID}.")
                    else:
                        logger.warning(f"Bot lacks 'Delete Messages' permission in channel {DATA_CHANNEL_ID}.")
                except (MessageIdInvalid, FloodWait, ChatAdminRequired) as e:
                    logger.warning(f"Could not delete channel message {message_id_in_channel}: {e}")
                except Exception as e:
                    logger.error(f"Failed to delete channel message {message_id_in_channel}: {e}", exc_info=True)
            return True
        return False
    except Exception as e:
        logger.error(f"Failed to delete broken video {video_uuid}: {e}", exc_info=True)
        return False

async def send_and_replace_message(client: Client, chat_id: int, message_id_to_edit_or_delete: int, new_message_type: str, video_data: dict = None, reply_markup: InlineKeyboardMarkup = None, text_content: str = None, force_new_message: bool = False, is_batch: bool = False, batch_id: str = None, batch_index: int = -1) -> tuple[bool, Message | str]:
    settings = settings_collection.find_one({'_id': 'settings'}) or {}
    protect_content_for_user = settings.get('protect_content', True)
    if new_message_type == "video" and not DATA_CHANNEL_ID:
        return False, "Video storage channel is not configured. Please contact an admin."

    caption_text = ""
    if new_message_type == "video" and video_data:
        current_position, total_videos = 0, 0
        if is_batch:
            _, current_position, total_videos = get_video_and_position_batch(batch_id, batch_index)
        else:
            is_saved_from_markup = any(b.callback_data and ('|' in b.callback_data) and bool(int(b.callback_data.split('|')[3])) for r in reply_markup.inline_keyboard for b in r if b.callback_data) if reply_markup else False
            _, current_position, total_videos = get_video_and_position(video_data['uuid'], video_data.get('category'), is_saved_from_markup, chat_id)
        custom_caption = video_data.get('custom_caption')
        category_caption = f"Category: {html.escape(video_data['category'])}" if video_data.get('category') else ""
        position_caption = f"#{current_position} of {total_videos}\n" if total_videos > 0 and current_position > 0 else ""
        caption_text = f"{position_caption}{custom_caption or category_caption}".strip()

    if force_new_message and message_id_to_edit_or_delete:
        try: await client.delete_messages(chat_id, message_id_to_edit_or_delete)
        except Exception as e: logger.warning(f"Failed to delete old message {message_id_to_edit_or_delete}: {e}")

    sent_message = None
    if new_message_type == "video" and video_data:
        if not force_new_message and message_id_to_edit_or_delete:
            try:
                sent_message = await client.edit_message_media(chat_id, message_id_to_edit_or_delete, InputMediaVideo(video_data['file_id'], caption=caption_text), reply_markup=reply_markup)
            except FileReferenceExpired:
                logger.warning(f"FileReferenceExpired on edit for video {video_data['uuid']}. Healing...")
                try:
                    healed_message = await client.get_messages(DATA_CHANNEL_ID, video_data['message_id'])
                    new_file_id = healed_message.video.file_id
                    media_collection.update_one({'uuid': video_data['uuid']}, {'$set': {'file_id': new_file_id}})
                    video_data['file_id'] = new_file_id
                    sent_message = await client.edit_message_media(chat_id, message_id_to_edit_or_delete, InputMediaVideo(new_file_id, caption=caption_text), reply_markup=reply_markup)
                except Exception as heal_e:
                    logger.error(f"Self-healing or retry-edit failed: {heal_e}. Sending new message.")
            except Exception as e:
                logger.warning(f"Editing message {message_id_to_edit_or_delete} failed: {e}. Sending new message.")
                if message_id_to_edit_or_delete:
                    try: await client.delete_messages(chat_id, message_id_to_edit_or_delete)
                    except Exception: pass
        if not sent_message:
            try:
                sent_message = await client.copy_message(chat_id, DATA_CHANNEL_ID, video_data.get('message_id'), caption=caption_text, protect_content=protect_content_for_user, reply_markup=reply_markup)
            except (ChatAdminRequired, MessageIdInvalid) as e:
                logger.warning(f"copy_message failed for {video_data['uuid']}: {e}. Triggering deletion.")
                await client.send_message(chat_id, "Oops! This video seems to be permanently unavailable and has been removed. Trying the next one... 🔄")
                await delete_broken_video_from_db_and_channel(client, video_data['uuid'], video_data.get('category'), video_data.get('sequence_number'), video_data.get('message_id'))
                return False, "Video was permanently unavailable."
            except Exception as e:
                logger.error(f"copy_message failed for {video_data['uuid']}: {e}. Falling back to send_video.", exc_info=True)
                try:
                    sent_message = await client.send_video(chat_id, video_data['file_id'], caption=caption_text, reply_markup=reply_markup, protect_content=protect_content_for_user)
                except FileReferenceExpired:
                    logger.warning(f"FileReferenceExpired on send_video for {video_data['uuid']}. Healing...")
                    try:
                        healed_message = await client.get_messages(DATA_CHANNEL_ID, video_data['message_id'])
                        new_file_id = healed_message.video.file_id
                        media_collection.update_one({'uuid': video_data['uuid']}, {'$set': {'file_id': new_file_id}})
                        video_data['file_id'] = new_file_id
                        sent_message = await client.send_video(chat_id, new_file_id, caption=caption_text, reply_markup=reply_markup, protect_content=protect_content_for_user)
                    except Exception as heal_e:
                        logger.error(f"Self-healing failed for video {video_data['uuid']}: {heal_e}", exc_info=True)
                        await client.send_message(chat_id, "Oops! This video is unavailable and has been removed. Trying the next one... 🔄")
                        await delete_broken_video_from_db_and_channel(client, video_data['uuid'], video_data.get('category'), video_data.get('sequence_number'), video_data.get('message_id'))
                        return False, "Video was unavailable and self-healing failed."
                except Exception as e2:
                    logger.error(f"Fallback send_video also failed for {video_data['uuid']}: {e2}. Video broken.", exc_info=True)
                    await client.send_message(chat_id, "Oops! This video is unavailable and has been removed. Trying the next one... 🔄")
                    await delete_broken_video_from_db_and_channel(client, video_data['uuid'], video_data.get('category'), video_data.get('sequence_number'), video_data.get('message_id'))
                    return False, "Video was unavailable by all methods."
    elif new_message_type == "text" and text_content:
        try:
            if not force_new_message and message_id_to_edit_or_delete:
                sent_message = await client.edit_message_text(chat_id, message_id_to_edit_or_delete, text_content, reply_markup=reply_markup)
            else:
                sent_message = await client.send_message(chat_id, text_content, reply_markup=reply_markup)
        except Exception:
            try: sent_message = await client.send_message(chat_id, text_content, reply_markup=reply_markup)
            except Exception as e2: return False, f"Failed to send message: {e2}"
    if isinstance(sent_message, Message):
        set_active_video_message(chat_id, sent_message.id, chat_id)
        return True, sent_message
    return False, "Failed to send or edit the message due to an unexpected error."

async def handle_token_refresh(user_id: int, ad_code: str) -> tuple[bool, str]:
    try:

# --- Rate Limiting ---
RATE_LIMIT_WINDOW, RATE_LIMIT_MAX = 120, 60
async def is_rate_limited(user_id: int) -> bool:
    if is_admin(user_id): return False
    now, window_start = datetime.utcnow(), datetime.utcnow() - timedelta(seconds=RATE_LIMIT_WINDOW)
    try:
        tokens_collection.update_one({'user_id': user_id}, {'$pull': {'rate_limits': {'timestamp': {'$lt': window_start}}}}, upsert=True)
        tokens_collection.update_one({'user_id': user_id}, {'$push': {'rate_limits': {'timestamp': now}}})
        result = tokens_collection.find_one({'user_id': user_id})
        current_requests = len(result.get('rate_limits', [])) if result else 0
        if current_requests > RATE_LIMIT_MAX:
            logger.warning(f"User {user_id} is rate limited. Requests: {current_requests}")
            return True
        return False
    except Exception as e:
        logger.error(f"Error in is_rate_limited for user {user_id}: {e}", exc_info=True)
        return False

async def verify_telegram_init_data(request: Request) -> dict:
    try:
        init_data_str = request.headers.get("X-Telegram-Init-Data")
        if not init_data_str: raise HTTPException(status_code=401, detail="Unauthorized: Missing Telegram Init Data")
        params = dict(parse_qs(init_data_str))
        hash_from_telegram = params.pop('hash', [None])[0]
        if not hash_from_telegram: raise HTTPException(status_code=401, detail="Unauthorized: Hash not found in Init Data")
        data_check_string = "\n".join(f"{k}={v[0]}" for k, v in sorted(params.items()))
        secret_key = hmac.new("WebAppData".encode(), config.BOT_TOKEN.encode(), hashlib.sha256).digest()
        calculated_hash = hmac.new(secret_key, data_check_string.encode(), hashlib.sha256).hexdigest()
        if calculated_hash != hash_from_telegram:
            logger.error(f"API call with invalid hash. Calculated: {calculated_hash}, Received: {hash_from_telegram}")
            raise HTTPException(status_code=403, detail="Forbidden: Invalid data signature")
        user_data_json = params.get('user', [None])[0]
        if not user_data_json: raise HTTPException(status_code=401, detail="Unauthorized: User data not found")
        user_data = json.loads(unquote(user_data_json))
        user_id = user_data.get('id')
        if not user_id: raise HTTPException(status_code=401, detail="Unauthorized: User ID not found")
        logger.info(f"API call verified for user_id: {user_id}")
        return {"user_id": user_id}
    except json.JSONDecodeError: raise HTTPException(status_code=400, detail="Bad Request: Invalid JSON")
    except Exception as e:
        logger.error(f"Error during initData verification: {e}", exc_info=True)

@fastapi_app.get("/stream/{video_uuid}")
async def stream_video(video_uuid: str, request: Request, token: str = None, download: bool = False):
    if not token: raise HTTPException(status_code=401, detail="Unauthorized: Missing stream token")
    try:
        decoded_payload = base64.urlsafe_b64decode(token + "==").decode()
        token_uuid, token_user_id_str, token_timestamp_str = decoded_payload.split(':')
        if get_current_time() - int(token_timestamp_str) > 3600: raise HTTPException(status_code=401, detail="Unauthorized: Expired stream token")
        if token_uuid != video_uuid: raise HTTPException(status_code=403, detail="Forbidden: Token mismatch")
        if not user_has_token(int(token_user_id_str)) and not is_premium_user(int(token_user_id_str)): raise HTTPException(status_code=403, detail="Forbidden: User access has expired")
    except Exception: raise HTTPException(status_code=401, detail="Unauthorized: Invalid stream token")
    video_doc = await async_media_collection.find_one({'uuid': video_uuid})
    if not video_doc: raise HTTPException(status_code=404, detail="Video not found")
    file_id = video_doc.get('file_id')
    input_file_location, file_size, _, file_type, file_name = decode_file_id(file_id)
    if not input_file_location: raise HTTPException(status_code=500, detail="Could not retrieve file information")
    mime_type = get_mime_type(file_name, file_type)
    range_header = request.headers.get("Range")
    start_byte, end_byte, status_code = 0, file_size - 1, 200
    headers = {"Accept-Ranges": "bytes", "Content-Type": mime_type, "Content-Disposition": f"{'attachment' if download else 'inline'}; filename=\"{file_name or video_uuid}\""}
    if range_header:
        match = re.search(r"bytes=(\d+)-(\d*)", range_header)
        if match:
            start_byte = int(match.group(1))
            if match.group(2): end_byte = int(match.group(2))
            status_code = 206
            headers["Content-Range"] = f"bytes {start_byte}-{end_byte}/{file_size}"
    content_length = end_byte - start_byte + 1
    headers["Content-Length"] = str(content_length)
    async def file_iterator():
        current_offset, bytes_to_send, CHUNK_SIZE = start_byte, content_length, 1024 * 1024
        nonlocal input_file_location
        while bytes_to_send > 0:
            try:
                chunk = await app.invoke(GetFile(location=input_file_location, offset=current_offset, limit=min(CHUNK_SIZE, bytes_to_send)))
                if not chunk.bytes: break
                yield chunk.bytes
                current_offset += len(chunk.bytes)
                bytes_to_send -= len(chunk.bytes)
            except FileReferenceExpired:
                logger.warning(f"FileReferenceExpired during stream for {video_uuid}. Healing...")
                healed_message = await app.get_messages(DATA_CHANNEL_ID, video_doc['message_id'])
                new_file_id = healed_message.video.file_id
                await async_media_collection.update_one({'uuid': video_uuid}, {'$set': {'file_id': new_file_id}})
                input_file_location, _, _, _, _ = decode_file_id(new_file_id)
                continue
            except FloodWait as e: await asyncio.sleep(e.value); continue
            except Exception as e: logger.error(f"Error streaming chunk for {video_uuid}: {e}"); break
    return StreamingResponse(file_iterator(), status_code=status_code, headers=headers)

@fastapi_app.post("/api/bookmark")
async def toggle_api_bookmark(request: BookmarkRequest, auth: dict = Depends(verify_telegram_init_data)):
    user_id, video_uuid = auth["user_id"], request.video_uuid
    user_doc = await async_users_collection.find_one({'user_id': user_id})
    if not user_doc: raise HTTPException(status_code=404, detail="User not found")
    bookmarked_videos = user_doc.get('bookmarked_videos', [])
    is_already_bookmarked = any(v['uuid'] == video_uuid for v in bookmarked_videos)
    if is_already_bookmarked:
        await async_users_collection.update_one({'user_id': user_id}, {'$pull': {'bookmarked_videos': {'uuid': video_uuid}}})
        return JSONResponse(content={"status": "removed"}, status_code=200)
    else:
        if not is_premium_user(user_id) and len(bookmarked_videos) >= config.FREE_USER_SAVE_LIMIT:
            raise HTTPException(status_code=403, detail=f"Bookmark limit of {config.FREE_USER_SAVE_LIMIT} reached.")
        video_data = await async_media_collection.find_one({'uuid': video_uuid})
        if not video_data: raise HTTPException(status_code=404, detail="Video to bookmark not found.")
        new_bookmark = {'uuid': video_uuid, 'bookmarked_at': datetime.utcnow(), 'category': video_data.get('category')}
        await async_users_collection.update_one({'user_id': user_id}, {'$push': {'bookmarked_videos': new_bookmark}})
        return JSONResponse(content={"status": "added"}, status_code=201)

    await broadcast_msg.edit_text(f"Broadcast Completed! 🎉\n\nTotal Users: {total_users}\nSuccessful: {success} ✅\nBlocked: {blocked} 🚫\nUnsuccessful: {failed} ❌")

async def process_batch_queue(user_id: int, client: Client):
    state = batch_add_state.get(user_id)
    if not state or not state.get('batch_mode'):
        if state: state['is_processing'] = False; return
    queue = state['video_queue']
    if not DATA_CHANNEL_ID: await client.send_message(user_id, "❌ Video storage channel not configured."); state['is_processing'] = False; return
    while queue and batch_add_state.get(user_id, {}).get('batch_mode'):
        message = queue.popleft()
        category = state['current_category']
        try:
            file_unique_id = message.video.file_unique_id
            if media_collection.find_one({"file_unique_id": file_unique_id}):
                await message.reply_text("⚠️ This video has already been added. Skipping."); continue
            sent_video_message = await client.send_video(DATA_CHANNEL_ID, message.video.file_id, caption=f"Added by admin {user_id} for category {category}")
            video_uuid = str(uuid.uuid4())
            next_sequence_number = state['next_sequence']
            video_data = {"uuid": video_uuid, "file_id": sent_video_message.video.file_id, "file_unique_id": file_unique_id, "category": category, "timestamp": get_current_time(), "sequence_number": next_sequence_number, "message_id": sent_video_message.id, "banned": False, "custom_caption": message.caption}
            media_collection.insert_one(video_data)
            state['next_sequence'] += 1
            state['videos_this_session'] += 1
            await message.reply_text(f"✅ File Added to <b>{html.escape(category)}</b>!\n🔢 Sequence: {next_sequence_number}\nProcessed this session: <b>{state['videos_this_session']}</b>")
            await asyncio.sleep(5)
        except FloodWait as e:
            await message.reply_text(f"⏳ FloodWait. Pausing for {e.value + 1} seconds.")
            queue.appendleft(message); await asyncio.sleep(e.value + 1)
        except Exception as e:
            logger.error(f"Error adding video from queue: {e}", exc_info=True)
            await message.reply("❌ Failed to add this video. Skipping.")
    if batch_add_state.get(user_id): batch_add_state[user_id]['is_processing'] = False
    if not batch_add_state.get(user_id, {}).get('video_queue'):
        await client.send_message(user_id, "✅ All queued videos have been processed. Send more or type /done.")
